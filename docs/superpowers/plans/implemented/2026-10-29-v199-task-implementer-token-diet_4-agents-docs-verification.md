"""Agent files, /task-brief and the docs name the handoff contract task_handoff.py exports (v199).

Renaming a section, a report prefix or a deliverable state on one side
without the other fails here.
"""
from __future__ import annotations

import importlib.util
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[2]
_SPEC = importlib.util.spec_from_file_location(
    "task_handoff", ROOT / "scripts" / "dev" / "task_handoff.py")
task_handoff = importlib.util.module_from_spec(_SPEC)
_SPEC.loader.exec_module(task_handoff)

AGENTS = ("task-implementer", "task-reviewer")


def _text(rel: str) -> str:
    return (ROOT / rel).read_text(encoding="utf-8")


def _agent(name: str) -> str:
    return _text(f".claude/agents/{name}.md")


@pytest.mark.parametrize("name", AGENTS)
def test_agent_names_every_handoff_section(name):
    body = _agent(name)
    missing = [s for s in task_handoff.SECTIONS if f"`## {s}`" not in body]
    assert not missing, missing


@pytest.mark.parametrize("name", AGENTS)
def test_agent_names_the_no_handoff_fallback(name):
    assert task_handoff.NO_HANDOFF_LINE in _agent(name)


def test_implementer_report_line_uses_the_exported_states():
    done, deviated = task_handoff.IMPLEMENTER_STATES
    prefix = task_handoff.REPORT_PREFIX
    assert f"{prefix} D1 {done} | D2 {done} | D3 {deviated}: <why>" in _agent("task-implementer")


def test_implementer_carries_the_handoff_ui_and_plan_rules():
    body = _agent("task-implementer")
    assert "task_handoff.py" in body
    assert "Never open a plan file" in body
    assert "Never edit the handoff file" in body
    assert "layout-audit.mjs" in body and "--clip <css selector>" in body
    assert f"at most {task_handoff.MAX_IMAGE_READS} image reads per task" in body.lower()
    assert "verification-before-completion" not in body


def test_reviewer_lines_use_the_exported_states():
    body = _agent("task-reviewer")
    done, missing, different = task_handoff.REVIEWER_STATES
    for state in task_handoff.REVIEWER_STATES:
        assert f"`D<n> {state}" in body, state
    assert f"D1 {done}\nD2 {missing} -- <what>\nD3 {different} -- <what>\n" in body


def test_reviewer_reads_the_handoff_not_the_report():
    body = _agent("task-reviewer")
    assert "never the implementer's report" in body
    assert "`[spec]` finding" in body
```

````markdown
---
name: task-implementer
description: "Implements exactly one plan task from its handoff file (task text, numbered deliverables, environment, trap preflight and code excerpts), test-first, in the worktree the handoff names, and returns a short report with one tick per deliverable. ## Rules

- **Read the handoff first, once, whole.** Trust its `## Code` excerpts
  before searching; look up only what it does not carry. Batch independent
  lookups into one call, and read files by line range, not whole.
- **Never open a plan file**, with no exception: the handoff carries the
  task text. If the handoff is missing something, return `BLOCKED:` — do not
  go looking.
- **No environment probing.** `## Environment` already answers the
  worktree, branch, test command, database, frontend wrapper, radon and
  attribution; do not re-check them. Never edit the handoff file.

  `testrun.py file` command — never the full suite, never `fast`; the plan's
  last task owns the full run.
- Every function you write or change stays below cyclomatic complexity 15:
  `python -m radon cc -s -n C <files>` must print nothing for your changes.
- Area skills (`no-lookahead`, `alert-surface`, `edge-module`,
  `schema-change`) fire on their own when you touch their area — follow them.
- Never `cd` in a Bash command (it breaks the repo hooks); use absolute paths
  or `git -C <path>`.
- **UI tasks** (any path under `frontend/`): run
  `frontend/scripts/layout-audit.mjs` first and work from its findings. Read
  a screenshot only for a named element, clipped with
  `browse.mjs shots --clip <css selector>`. At most 4 image reads per task;
  needing more is `BLOCKED:` with the reason, never a silent overrun.
- Commit when the task's steps say so, with the task's message, ending with
  the attribution line from `## Environment`.
- Before you report, check each deliverable against your own commits. A
  deliverable you changed or skipped is `deviated: <why>`, never `done`.

Return **only** this, nothing else:

```
STATUS: DONE | BLOCKED: <one question>
COMMITS: <sha> <subject>   (one line each)
FILES: <path>              (one line each)
TESTS: <testrun.py verdict line, verbatim, from a run after your last edit>
RADON: clean | <function: grade>
DELIVERABLES: D1 done | D2 done | D3 deviated: <why>
NOTES: <= 3 lines, only what the reviewer must know
```

`DELIVERABLES:` names every `Dn` and `Dn.k` of the handoff, in order, each

Read-only. Use after task-implementer so review runs on Sonnet and only findings reach Opus."
tools: Bash, Read, Grep, Glob, Skill

You review **one** task. You get the task id, the absolute path of its

## Order

1. **Deliverables.** For each `Dn` and `Dn.k` in `## Deliverables`, in
   order, check the diff (`git -C <wt> diff <range>`) and the task text in
   `## Task`: `done`, `missing` or `different`. Every `missing` or
   `different` is also a `[spec]` finding, and so is anything in the diff
   that no deliverable or step asks for.
2. **Correctness.** Lookahead (preloaded skill), off-by-one on bar indices,
   silent no-op shims (`docs/claude/known-traps.md`), untested branches.
3. **Complexity.** `python -m radon cc -s -n C <changed .py files>` — any
   function at C or worse that the diff created or made worse is a finding.
4. **Tests.** `python scripts/dev/testrun.py file <each changed test file>`;
   a failure is a finding with the verdict line quoted.

Dispatched with a pasted brief and no handoff file: check every step of the
brief in step 1 instead, and print `DELIVERABLES: n/a (no handoff)` in place
of the `D` lines.

The preloaded `senior-engineer` skill is the checklist for steps 2 to 4; a
diff that matches one of its red flags is always a finding. You stay on
sonnet whatever model implemented the task.

Do not report style preferences, and do not suggest refactors outside the
diff. A finding you cannot tie to a line is not a finding.

## Return shape

```
VERDICT: CLEAN | FINDINGS | BLOCKED: <one question>
D1 done
D2 missing -- <what>
D3 different -- <what>
1. [spec|correctness|complexity|test] <file>:<line> -- <one sentence>
   fix: <one sentence>
...
```

One `D` line per deliverable, in handoff order, before any finding:
`D<n> done`, `D<n> missing -- <what>` or `D<n> different -- <what>`
(sub-items as `D<n>.<k>`). A `missing` or `different` line always has its
`[spec]` finding below. At most 10 findings, most severe first.
````
