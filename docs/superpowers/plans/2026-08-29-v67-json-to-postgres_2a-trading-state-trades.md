## Parallelisation

Two groups, taken directly from the spec, plus the sequential edges that make
them safe:

- **Sequential: P2-01 and P2-02 before everything else in this part.** P2-02
  introduces `scripts/db/import_common.py`, which every other importer here
  consumes.
- **Group 2a (parallel):** the trades chain (P2-03…P2-06), the plans chain
  (P2-07…P2-10), the starred chain (P2-11, P2-12). One module each —
  `tracking/performance.py`, `planning/plan_store.py`, `commands/views.py` — no
  shared file.
- **Group 2b (parallel):** account (P2-13…P2-16), journal (P2-17, P2-18), state
  (P2-19, P2-20), watchlist (P2-21, P2-22). One module each.
- **Sequential within every chain.** A store's repository, its importer, its
  dual-write wiring and its read flip are a strict chain: each consumes the
  previous step's schema or stage behaviour.
- **`schema.py` is the one shared file in this part.** Every chain adds its
  table there. Two chains editing it concurrently is exactly the silent
  overwrite `docs/claude/document-conventions.md` warns about, so **all table
  definitions land in P2-01**, before either group starts. No later task in
  Part 2 edits `schema.py`.

```python
"""Trades. The queries TradeLog's public methods need, and nothing else."""
from __future__ import annotations

import sqlalchemy as sa

from swingbot.core.db.repositories.base import Repository
from swingbot.core.db.schema import trades

# Every status that is not "open". Kept as an explicit open-test rather than a
# closed-list so a status this repo adds later (v58 added "partial") counts as
# closed only when someone says so.
OPEN_STATUS = "open"


class TradeRepository(Repository):
    def __init__(self):
        super().__init__(trades, key="trade_id")

    def open_trades(self, *, conn=None) -> list[dict]:
        return self.list_all(conn=conn, where=trades.c.status == OPEN_STATUS,
                             order_by=trades.c.opened_at.desc())

    def closed_trades(self, *, conn=None, limit: int | None = None) -> list[dict]:
        return self.list_all(conn=conn, where=trades.c.status != OPEN_STATUS,
                             order_by=trades.c.closed_at.desc(), limit=limit)

    def by_ticker(self, ticker: str, *, conn=None) -> list[dict]:
        return self.list_all(
            conn=conn,
            where=sa.func.upper(trades.c.ticker) == (ticker or "").upper(),
            order_by=trades.c.opened_at.desc(),
        )

    def open_for_ticker(self, ticker: str, *, conn=None) -> dict | None:
        rows = self.list_all(
            conn=conn,
            where=sa.and_(sa.func.upper(trades.c.ticker) == (ticker or "").upper(),
                          trades.c.status == OPEN_STATUS),
            order_by=trades.c.opened_at.desc(), limit=1,
        )
        return rows[0] if rows else None

    def has_open(self, ticker: str, strategy: str, horizon: str,
                 direction: str, *, conn=None) -> bool:
        return self.count(conn=conn, where=sa.and_(
            sa.func.upper(trades.c.ticker) == (ticker or "").upper(),
            trades.c.strategy == strategy,
            trades.c.horizon == horizon,
            trades.c.direction == direction,
            trades.c.status == OPEN_STATUS,
        )) > 0

    def by_plan_id(self, plan_id: str, *, conn=None) -> dict | None:
        rows = self.list_all(conn=conn,
                             where=trades.c.doc["plan_id"].astext == plan_id,
                             limit=1)
        return rows[0] if rows else None

    def clear(self, *, status: str | None = None, conn=None) -> int:
        stmt = sa.delete(trades)
        if status == "open":
            stmt = stmt.where(trades.c.status == OPEN_STATUS)
        elif status == "closed":
            stmt = stmt.where(trades.c.status != OPEN_STATUS)
        with self._tx(conn) as c:
            return c.execute(stmt).rowcount


_repo: TradeRepository | None = None


def trades_repo() -> TradeRepository:
    global _repo
    if _repo is None:
        _repo = TradeRepository()
    return _repo
```

"""The import verifier. This is success criterion 2, so it is tested first."""
import pytest

from scripts.db.import_common import ImportReport, compare, record_checksum


def test_checksum_ignores_key_order():
    assert record_checksum({"a": 1, "b": 2}) == record_checksum({"b": 2, "a": 1})


def test_checksum_ignores_nested_key_order():
    assert (record_checksum({"x": {"a": 1, "b": 2}})
            == record_checksum({"x": {"b": 2, "a": 1}}))


def test_checksum_distinguishes_a_changed_value():
    assert record_checksum({"a": 1}) != record_checksum({"a": 2})


def test_checksum_distinguishes_int_from_string():
    # A field that arrives from JSONB as "4" instead of 4 is a real import bug,
    # not noise. Unlike the dual-write comparator, this one does not forgive.
    assert record_checksum({"a": 1}) != record_checksum({"a": "1"})


def test_checksum_survives_a_non_json_value():
    import datetime as dt
    assert record_checksum({"t": dt.datetime(2026, 1, 2)})  # does not raise


def test_a_clean_import_reports_ok():
    rows = [{"trade_id": "T1", "a": 1}, {"trade_id": "T2", "a": 2}]
    report = compare(rows, list(rows), key="trade_id")
    assert report.ok
    assert report.source_count == report.imported_count == 2
    assert not (report.missing or report.extra or report.mismatched)


def test_a_missing_record_is_named():
    src = [{"trade_id": "T1"}, {"trade_id": "T2"}]
    report = compare(src, [{"trade_id": "T1"}], key="trade_id")
    assert not report.ok
    assert report.missing == ["T2"]


def test_an_extra_record_is_named():
    report = compare([{"trade_id": "T1"}],
                     [{"trade_id": "T1"}, {"trade_id": "T9"}], key="trade_id")
    assert not report.ok
    assert report.extra == ["T9"]


def test_a_changed_field_is_named():
    report = compare([{"trade_id": "T1", "entry": 100.0}],
                     [{"trade_id": "T1", "entry": 101.0}], key="trade_id")
    assert not report.ok
    assert report.mismatched == ["T1"]


def test_a_duplicate_key_in_the_source_is_fatal():
    with pytest.raises(ValueError, match="duplicate"):
        compare([{"trade_id": "T1"}, {"trade_id": "T1"}], [], key="trade_id")


def test_render_names_every_failure_category():
    report = compare([{"trade_id": "T1"}, {"trade_id": "T2", "a": 1}],
                     [{"trade_id": "T2", "a": 2}, {"trade_id": "T9"}],
                     key="trade_id")
    text = report.render()
    assert "T1" in text and "T2" in text and "T9" in text
    assert "MISMATCH" in text

"""Shared import verification: row counts and per-record checksums.

Every importer in this plan runs a dry run first. The data being imported is
irreplaceable -- there is no second copy of the VM's trade history -- so the
verification is not optional and is not a flag someone can forget.
"""
from __future__ import annotations

import argparse
import hashlib
import json
from dataclasses import dataclass, field


def record_checksum(record: dict) -> str:
    """sha256 over a canonical JSON form.

    sort_keys makes it key-order independent; default=str keeps a datetime or
    Decimal from raising rather than silently being skipped. Types are NOT
    coerced -- 1 and "1" hash differently on purpose, because a field arriving
    from JSONB as a string is exactly the import bug this exists to catch.
    """
    blob = json.dumps(record, sort_keys=True, default=str, separators=(",", ":"))
    return hashlib.sha256(blob.encode("utf-8")).hexdigest()


@dataclass
class ImportReport:
    source_count: int = 0
    imported_count: int = 0
    missing: list[str] = field(default_factory=list)
    extra: list[str] = field(default_factory=list)
    mismatched: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return (self.source_count == self.imported_count
                and not self.missing and not self.extra and not self.mismatched)

    def render(self) -> str:
        lines = [
            f"  source records : {self.source_count}",
            f"  imported rows  : {self.imported_count}",
        ]
        for label, ids in (("MISSING (in source, not imported)", self.missing),
                           ("EXTRA (imported, not in source)", self.extra),
                           ("MISMATCH (checksum differs)", self.mismatched)):
            if ids:
                shown = ", ".join(ids[:20])
                more = f" (+{len(ids) - 20} more)" if len(ids) > 20 else ""
                lines.append(f"  {label}: {shown}{more}")
        lines.append("  VERDICT: " + ("OK" if self.ok else "FAILED"))
        return "\n".join(lines)


def _index(rows: list[dict], key: str, label: str) -> dict[str, dict]:
    out: dict[str, dict] = {}
    for row in rows:
        k = str(row.get(key))
        if k in out:
            raise ValueError(f"duplicate {key}={k!r} in {label}")
        out[k] = row
    return out


def compare(source: list[dict], imported: list[dict], key: str) -> ImportReport:
    src = _index(source, key, "source")
    dst = _index(imported, key, "imported")
    report = ImportReport(source_count=len(src), imported_count=len(dst))
    report.missing = sorted(set(src) - set(dst))
    report.extra = sorted(set(dst) - set(src))
    report.mismatched = sorted(
        k for k in set(src) & set(dst)
        if record_checksum(src[k]) != record_checksum(dst[k])
    )
    return report


def run_import(argv, *, load_source, write_one, repo, key: str,
               name: str) -> int:
    """The CLI every importer shares.

    --dry-run  read the source, report what WOULD be written, touch nothing
    (default)  write, then re-read and verify; a failed verification is a
               non-zero exit AND leaves the rows in place for inspection --
               rolling back would destroy the evidence of what went wrong.
    """
    ap = argparse.ArgumentParser(description=f"Import {name} into Postgres")
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--source", help="path to the JSON file (default: data/)")
    args = ap.parse_args(argv)

    source = load_source(args.source)
    print(f"[{name}] {len(source)} record(s) in source", flush=True)

    if args.dry_run:
        existing = repo.count()
        print(f"[{name}] DRY RUN -- would write {len(source)} row(s); "
              f"table currently holds {existing}")
        return 0

    for i, record in enumerate(source, 1):
        write_one(repo, record)
        if i % 100 == 0 or i == len(source):
            print(f"[{name}] {i}/{len(source)} written", flush=True)

    imported = repo.list_all()
    report = compare(source, imported, key=key)
    print(f"[{name}] verification:")
    print(report.render())
    return 0 if report.ok else 1
```

The per-unit progress print is not decoration — `docs/claude/working-conventions.md`
requires it for anything that can run longer than a couple of minutes, and a
production trades import is exactly that.

    def _db_upsert(self, trade: dict) -> None:
        """Mirror one trade into Postgres. No-op below the dual stage.

        Never wrapped in try/except: at dual and db stages a write that cannot
        reach the database raises, on purpose. A dual stage that swallowed
        failures would report a clean parity run against a table nothing was
        ever written to, which is worse than no parity run at all.
        """
        from swingbot.core.db import stages
        if not stages.writes_db("trades"):
            return
        from swingbot.core.db.repositories.trades import trades_repo
        trades_repo().upsert(trade)

    def _db_delete(self, trade_id: str) -> None:
        from swingbot.core.db import stages
        if not stages.writes_db("trades"):
            return
        from swingbot.core.db.repositories.trades import trades_repo
        trades_repo().delete(trade_id)

    def _persist(self, trade: dict | None = None) -> None:
        """Write through to whichever backends this stage uses.

        `trade` is the single record that changed, when the caller knows it --
        that is what lets the db side issue one row write instead of rewriting
        the world, which is the whole point of the migration. Callers that
        genuinely changed many records (update_open_trades) pass None and get
        a full mirror.
        """
        from swingbot.core.db import stages
        if stages.writes_json("trades"):
            self._save()
        if not stages.writes_db("trades"):
            return
        if trade is not None:
            self._db_upsert(trade)
        else:
            for t in self._trades:
                self._db_upsert(t)
```

`account.py` at module load, and a module-level `swingbot.core.db` import here
would drag SQLAlchemy into every process that touches an account config,
including the ones that never open a database.
