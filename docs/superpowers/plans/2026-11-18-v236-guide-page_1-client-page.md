```ts
import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { firstValueFrom } from 'rxjs';

import { ApiClient } from '../api/api-client';
import { ApiError } from '../api/api-error';
import { GuideChapter, GuideChapterSummary } from '../api/models';
import { routeRequest } from '../routing/route-request';

export type GuideStatus = 'idle' | 'loading' | 'ready' | 'error';
export type GuideChapterStatus = 'loading' | 'ready' | 'missing' | 'error';

interface GuideSlice {
  chapters: GuideChapterSummary[];
  bodies: Record<string, GuideChapter>;
  status: GuideStatus;
  error: string | null;
  chapterStatus: Record<string, GuideChapterStatus>;
}

const UNAVAILABLE = 'The admin is not responding — the guide is unavailable.';

/**
 * The Guide page's data (v235): the chapter index once per visit, and each
 * chapter body the first time it is opened.
 *
 * Chapters change only with a deploy, so nothing refetches on an event. The
 * index carries every heading, so the table of contents and heading search
 * work before any body is fetched; bodies stay in `bodies` so search can
 * cover the text of every chapter the reader has opened.
 */
export const GuideStore = signalStore(
  withState<GuideSlice>({
    chapters: [], bodies: {}, status: 'idle', error: null, chapterStatus: {},
  }),
  withComputed(({ chapters }) => ({
    /** The chapter `/insights/guide` opens on: the first in reading order. */
    firstSlug: computed<string | null>(() => chapters()[0]?.slug ?? null),
  })),
  withMethods((store, api = inject(ApiClient)) => {
    let pending: Promise<void> | null = null;
    const setChapter = (slug: string, status: GuideChapterStatus) =>
      patchState(store, { chapterStatus: { ...store.chapterStatus(), [slug]: status } });
    const index = () => routeRequest(api.guide(), {
      start: () => patchState(store, { status: 'loading' }),
      next: ({ chapters }) => patchState(store, { chapters, status: 'ready', error: null }),
      error: (error) => patchState(store, {
        status: 'error',
        error: error.code === 'unavailable' ? UNAVAILABLE : error.message,
      }),
    });
    return {
      /** For the route resolver: resolves once the index has landed or
       *  failed; a later call after a failure fetches again. An auth
       *  failure rejects, so `startRoute` can expire the session. */
      ensureLoaded(): Promise<void> {
        if (store.status() === 'ready') return Promise.resolve();
        pending ??= firstValueFrom(index(), { defaultValue: undefined })
          .finally(() => { pending = null; });
        return pending;
      },
      loadChapter(slug: string): void {
        const now = store.chapterStatus()[slug];
        if (now === 'loading' || now === 'ready') return;
        setChapter(slug, 'loading');
        api.guideChapter(slug).subscribe({
          next: (chapter) => {
            patchState(store, { bodies: { ...store.bodies(), [slug]: chapter } });
            setChapter(slug, 'ready');
          },
          error: (error: unknown) => setChapter(
            slug, error instanceof ApiError && error.code === 'not_found' ? 'missing' : 'error'),
        });
      },
    };
  }),
);
```

```ts
import { ChangeDetectionStrategy, Component, computed, input } from '@angular/core';
import { RouterLink } from '@angular/router';

/** Where a `?` link lands: a chapter, and optionally one heading in it. */
export interface GuideTarget {
  slug: string;
  anchor: string | null;
}

/**
 * `'<slug>#<anchor>'` or `'<slug>'`. The anchor is the server's (the
 * browser never derives one); `guide-anchors.spec.ts` checks every literal
 * target in the app against the generated fixture, so a renamed heading
 * fails CI instead of leaving a link that lands at the top of a chapter.
 */
export function parseGuideTarget(target: string): GuideTarget {
  const [head, ...rest] = target.split('#');
  const slug = head.trim();
  if (slug === '') throw new Error(`Guide target "${target}" names no chapter`);
  const anchor = rest.join('#').trim();
  return { slug, anchor: anchor === '' ? null : anchor };
}

/**
 * The guide's contextual help (spec v215 § Page behaviour): a small `?`
 * beside an alert field or a metric that opens the matching guide section.
 *
 * The glyph is text-sized so it never changes the line it sits on; the
 * touch target is the shared `.sb-hit` box (44px on phone and tablet,
 * 24px on desktop), centred on the glyph and out of layout, so the row
 * keeps its height at every width (the lesson of commit 2a11a416).
 */
@Component({
  selector: 'sb-guide-link',
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [RouterLink],
  template: `
    <a class="guide-link sb-hit"
       [routerLink]="['/insights/guide', parsed().slug]"
       [fragment]="parsed().anchor ?? undefined"
       [attr.aria-label]="label()"
       [attr.title]="label()">?</a>
  `,
  styles: `
    :host { display: inline-flex; vertical-align: middle; }
    .guide-link {
      display: inline-flex;
      align-items: center;
      justify-content: center;
      width: 1.4em;
      height: 1.4em;
      border: 1px solid var(--border-strong);
      border-radius: 50%;
      color: var(--text-muted);
      font-size: var(--text-micro);
      font-weight: 600;
      line-height: 1;
      text-decoration: none;
      transition: color var(--transition), border-color var(--transition);
    }
    .guide-link:hover { color: var(--text); border-color: var(--text-muted); }
    .guide-link:focus-visible { outline: 1px solid var(--accent); outline-offset: 1px; }
  `,
})
export class GuideLinkComponent {
  /** `'<slug>#<anchor>'` or `'<slug>'`. Always a literal in templates. */
  readonly target = input.required<string>();
  /** The accessible name; the visible glyph is only "?". */
  readonly label = input<string>('Open the guide');

  protected readonly parsed = computed(() => parseGuideTarget(this.target()));
}
```

```ts
import { readdirSync, readFileSync } from 'node:fs';
import { join, relative } from 'node:path';
import { describe, expect, it } from 'vitest';

import { parseGuideTarget } from '../../ui/guide-link';

/**
 * Spec v215 freshness layer 2: every `sb-guide-link` target in the app names
 * a chapter and heading that exist. The anchors come from
 * guide-anchors.json, which `python scripts/dev/guide_anchors.py` generates
 * from docs/guide/ and the backend test (`tests/scripts/test_guide_anchors.py`)
 * checks against the chapters, so the two sides cannot drift apart.
 */

interface AnchorFixture { chapters: Record<string, string[]> }
interface Placement { file: string; target: string }

const APP = join(process.cwd(), 'src/app');
const FIXTURE: AnchorFixture = JSON.parse(
  readFileSync(join(APP, 'workspaces/guide/guide-anchors.json'), 'utf8'));

const LITERAL = /<sb-guide-link\b[^>]*?\starget="([^"]*)"/g;
const BOUND = /<sb-guide-link\b[^>]*?\[target\]=/g;

function sources(dir: string = APP): { file: string; source: string }[] {
  return readdirSync(dir, { withFileTypes: true }).flatMap((entry) => {
    const full = join(dir, entry.name);
    if (entry.isDirectory()) return sources(full);
    if (!/\.(ts|html)$/.test(entry.name) || entry.name.endsWith('.spec.ts')) return [];
    return [{ file: relative(APP, full).replace(/\\/g, '/'), source: readFileSync(full, 'utf8') }];
  });
}

function placements(files: { file: string; source: string }[]): Placement[] {
  return files.flatMap(({ file, source }) =>
    [...source.matchAll(LITERAL)].map((m) => ({ file, target: m[1] })));
}

function targetProblems(found: Placement[], fixture: AnchorFixture): string[] {
  return found.flatMap(({ file, target }) => {
    let slug: string;
    let anchor: string | null;
    try {
      ({ slug, anchor } = parseGuideTarget(target));
    } catch {
      return [`${file}: "${target}" names no chapter`];
    }
    const anchors = fixture.chapters[slug];
    if (!anchors) return [`${file}: "${target}" names no chapter "${slug}"`];
    if (anchor !== null && !anchors.includes(anchor)) {
      return [`${file}: "${target}" names no heading "${anchor}" in "${slug}"`];
    }
    return [];
  });
}

describe('guide-anchors.json', () => {
  it('is the generated shape: chapters keyed by slug, anchors as strings', () => {
    expect(Object.keys(FIXTURE)).toEqual(['chapters']);
    for (const [slug, anchors] of Object.entries(FIXTURE.chapters)) {
      expect(slug).toMatch(/^[a-z0-9-]+$/);
      expect(anchors.every((a) => typeof a === 'string' && a !== '')).toBe(true);
      expect(new Set(anchors).size).toBe(anchors.length);
    }
  });
});

describe('the anchor check itself', () => {
  const fixture: AnchorFixture = { chapters: { glossary: ['r-multiple'], pipeline: [] } };
  const at = (target: string) => [{ file: 'x.ts', target }];

  it('accepts a known chapter, alone or with a known heading', () => {
    expect(targetProblems(at('glossary'), fixture)).toEqual([]);
    expect(targetProblems(at('glossary#r-multiple'), fixture)).toEqual([]);
    expect(targetProblems(at('pipeline'), fixture)).toEqual([]);
  });

  it('names an unknown chapter, an unknown heading and an empty slug', () => {
    expect(targetProblems(at('nope'), fixture)).toEqual(['x.ts: "nope" names no chapter "nope"']);
    expect(targetProblems(at('glossary#nope'), fixture))
      .toEqual(['x.ts: "glossary#nope" names no heading "nope" in "glossary"']);
    expect(targetProblems(at('#r-multiple'), fixture)).toEqual(['x.ts: "#r-multiple" names no chapter']);
  });

  it('finds literal targets, with other attributes before them', () => {
    const files = [{ file: 'a.ts', source: '<sb-guide-link label="x" target="glossary#r-multiple" />' }];
    expect(placements(files)).toEqual([{ file: 'a.ts', target: 'glossary#r-multiple' }]);
  });
});

describe('every sb-guide-link in the app (v215 freshness layer 2)', () => {
  const files = sources();

  it('reads the app sources', () => {
    expect(files.some(({ file }) => file === 'ui/guide-link.ts')).toBe(true);
  });

  it('uses a literal target, so this spec can check it', () => {
    const bound = files
      .filter(({ source }) => { BOUND.lastIndex = 0; return BOUND.test(source); })
      .map(({ file }) => file);
    expect(bound).toEqual([]);
  });

  it('names an existing chapter and heading', () => {
    expect(targetProblems(placements(files), FIXTURE)).toEqual([]);
  });
});
```

`BOUND` is a global regex reused across files, so `lastIndex` is reset before each `.test()`.
