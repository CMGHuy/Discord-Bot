"""Alembic environment.

Reads the URL from swingbot.config and the target schema from
swingbot.core.db.schema, so a migration can never disagree with the running
application about either.
"""
from logging.config import fileConfig

from alembic import context

from swingbot import config as app_config
from swingbot.core.db.engine import get_engine
from swingbot.core.db.schema import METADATA

cfg = context.config
if cfg.config_file_name is not None:
    fileConfig(cfg.config_file_name)

target_metadata = METADATA


def run_migrations_offline() -> None:
    context.configure(
        url=app_config.DATABASE_URL,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    with get_engine().connect() as connection:
        context.configure(connection=connection, target_metadata=target_metadata)
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

def test_migrations_produce_exactly_the_declared_schema(db_engine_empty):
    """Upgrade a virgin database to head, then ask Alembic whether the result
    still differs from METADATA. Any output means someone edited schema.py
    without writing the revision that matches it."""
    from alembic.autogenerate import compare_metadata
    from alembic.migration import MigrationContext
    from alembic.command import upgrade

    from swingbot.core.db.schema import METADATA

    cfg = Config(str(REPO / "alembic.ini"))
    upgrade(cfg, "head")

    with db_engine_empty.connect() as conn:
        diff = compare_metadata(MigrationContext.configure(conn), METADATA)

    # Indexes on expressions (trades_ticker_opened_idx uses `opened_at DESC`)
    # are reported as differences by every alembic version this repo has run
    # against, because it cannot round-trip the expression. Filter those and
    # assert on everything else.
    real = [d for d in diff
            if not (isinstance(d, tuple) and str(d[0]).endswith("_index")
                    and "opened_idx" in str(d))]
    assert real == [], f"schema.py and the migrations disagree: {real}"
```

```python
"""Real-Postgres fixtures.

Isolation is per-test transaction rollback, not schema-per-test: creating the
schema once and rolling back each test is the difference between a database
tier that costs seconds and one that costs minutes. The exception is anything
that needs a COMMIT -- NOTIFY only fires on commit -- and those tests use
`db_committed` and truncate afterwards.
"""
import os

import pytest
import sqlalchemy as sa

from swingbot.core.db.schema import METADATA

DEFAULT_TEST_URL = "postgresql+psycopg://swingbot:swingbot@localhost:55432/swingbot_test"


def test_database_url() -> str:
    return os.getenv("TEST_DATABASE_URL", DEFAULT_TEST_URL)


def _connect_or_skip(url: str) -> sa.Engine:
    engine = sa.create_engine(url, future=True, pool_pre_ping=True)
    try:
        with engine.connect() as conn:
            conn.execute(sa.text("select 1"))
    except Exception as exc:  # noqa: BLE001 -- any connection failure, same advice
        engine.dispose()
        pytest.skip(
            f"test database unreachable at {url}: {exc}\n"
            f"Start it with: docker compose --profile test up -d db-test"
        )
    return engine


@pytest.fixture(scope="session")
def db_engine_empty():
    """An engine against a test database with NO schema. Migration tests
    build the schema themselves and must not find it already there."""
    engine = _connect_or_skip(test_database_url())
    with engine.begin() as conn:
        conn.execute(sa.text("DROP SCHEMA public CASCADE; CREATE SCHEMA public;"))
    yield engine
    engine.dispose()


@pytest.fixture(scope="session")
def db_engine():
    """An engine with the full declared schema created once for the session."""
    engine = _connect_or_skip(test_database_url())
    with engine.begin() as conn:
        conn.execute(sa.text("DROP SCHEMA public CASCADE; CREATE SCHEMA public;"))
    METADATA.create_all(engine)
    yield engine
    engine.dispose()


@pytest.fixture
def db_conn(db_engine):
    """A connection inside a transaction that is always rolled back.

    Anything the test writes is invisible to the next test and never
    committed, so tests can run in any order and leave nothing behind.
    """
    conn = db_engine.connect()
    trans = conn.begin()
    try:
        yield conn
    finally:
        trans.rollback()
        conn.close()


@pytest.fixture
def db_committed(db_engine):
    """A connection that COMMITS, for the few tests that need it (NOTIFY only
    fires on commit). Truncates every table afterwards -- slower than rollback,
    which is exactly why it is not the default."""
    conn = db_engine.connect()
    try:
        yield conn
    finally:
        conn.rollback()
        names = ", ".join(t.name for t in METADATA.sorted_tables)
        if names:
            with conn.begin():
                conn.execute(sa.text(f"TRUNCATE {names} RESTART IDENTITY CASCADE"))
        conn.close()
```

**Two fixtures that both reset the schema cannot be used in the same session**
— `db_engine_empty` drops what `db_engine` created. Keep migration tests
(`tests/db/test_migrations.py`) as the only user of `db_engine_empty`, and run
them in their own file; both are session-scoped, so pytest tears one down
before the other is built only if they are never requested together. If a
future test needs both, split the file.

quietly order-dependent."""
import sqlalchemy as sa

from swingbot.core.db.schema import trades

ROW = dict(trade_id="HARNESS-1", ticker="AAPL", strategy="RSI", horizon="2w",
           direction="LONG", status="open",
           opened_at="2026-01-02T15:00:00+00:00")


def test_a_write_is_visible_inside_its_own_test(db_conn):
    db_conn.execute(sa.insert(trades).values(**ROW))
    got = db_conn.execute(
        sa.select(trades.c.ticker).where(trades.c.trade_id == "HARNESS-1")
    ).scalar_one()
    assert got == "AAPL"


def test_the_previous_write_was_rolled_back(db_conn):
    count = db_conn.execute(
        sa.select(sa.func.count()).select_from(trades)
        .where(trades.c.trade_id == "HARNESS-1")
    ).scalar_one()
    assert count == 0, "rollback isolation is broken -- tests are order-dependent"


def test_doc_defaults_to_an_empty_object(db_conn):
    db_conn.execute(sa.insert(trades).values(**ROW))
    doc = db_conn.execute(
        sa.select(trades.c.doc).where(trades.c.trade_id == "HARNESS-1")
    ).scalar_one()
    assert doc == {}

    spec.loader.exec_module(mod)

    def refuse(*_a, **_kw):
        raise OSError("connection refused")
    monkeypatch.setattr(socket, "create_connection", refuse)

    assert "docker compose --profile test up -d db-test" in msg

def db_preflight() -> str | None:
    """Return a warning if the test database is unreachable, else None.

    A missing test database otherwise surfaces as a wall of connection errors
    from every tests/db/ test at once, which is the single least useful thing
    an agent can be handed.
    """
    import socket
    import urllib.parse

    url = os.getenv("TEST_DATABASE_URL", TEST_DB_URL_DEFAULT)
    parsed = urllib.parse.urlparse(url)
    host, port = parsed.hostname or "localhost", parsed.port or 5432
    try:
        with socket.create_connection((host, port), timeout=1.5):
            return None
    except OSError as exc:
        return (
            f"WARNING: test database unreachable at {host}:{port} ({exc}).\n"
            f"  Database tests will SKIP, not fail. Start it with:\n"
            f"    docker compose --profile test up -d db-test"
        )

### Task P1-09: The generic repository

One base class every store's repository subclasses. It owns the codec calls and
the connection handling, so a per-store repository is a table, a promoted tuple,
a key column — and nothing else.

The `update()` semantics are the point of the whole plan: it patches
`doc = doc || :patch` in SQL rather than rewriting a whole record, which is what
makes two concurrent writers safe. P1-14 proves it.

**Files:**
- Create: `swingbot/core/db/repositories/__init__.py`
- Create: `swingbot/core/db/repositories/base.py`
- Test: `tests/db/test_repository_base.py`

**Interfaces:**
- Consumes: `split_doc`/`merge_doc` (P1-01), `get_engine` (P1-02),
  `promoted_for`/`trades` (P1-04), `db_conn` (P1-07).
- Produces `class Repository`:
  - `__init__(self, table: sa.Table, key: str)`
  - `insert(record: dict, *, conn=None) -> dict`
  - `upsert(record: dict, *, conn=None) -> dict`
  - `get(key_value, *, conn=None) -> dict | None`
  - `list_all(*, conn=None, where=None, order_by=None, limit=None) -> list[dict]`

```python
"""The generic repository every store's repository subclasses.

`patch()` is the reason this migration exists. It issues
`UPDATE ... SET doc = doc || :patch WHERE key = :key`, so two processes
updating different fields of the same record both keep their write. The JSON
stores it replaces serialise an entire in-memory list over the whole file on
every write, which is why a stale reader could erase every trade logged
elsewhere in the meantime.
"""
from __future__ import annotations

from contextlib import contextmanager
from typing import Any, Iterator

import sqlalchemy as sa
from sqlalchemy.dialects.postgresql import insert as pg_insert

from swingbot.core.db.codec import merge_doc, split_doc
from swingbot.core.db.engine import get_engine
from swingbot.core.db.schema import promoted_for


class Repository:
    """CRUD over one table, in flat dicts.

    Every method takes an optional `conn`. Passing one joins the caller's
    transaction (which is how the test harness rolls everything back);
    omitting it opens and commits its own.
    """

    def __init__(self, table: sa.Table, key: str):
        self.table = table
        self.key = key
        self.promoted = promoted_for(table.name)
        self.key_col = table.c[key]

    # -- connection handling -------------------------------------------------

    @contextmanager
    def _tx(self, conn: sa.Connection | None) -> Iterator[sa.Connection]:
        if conn is not None:
            yield conn
        else:
            with get_engine().begin() as owned:
                yield owned

    # -- helpers -------------------------------------------------------------

    def _row_to_record(self, row) -> dict:
        return merge_doc(row._mapping, self.promoted)

    def _values(self, record: dict) -> dict:
        columns, doc = split_doc(record, self.promoted)
        return {**columns, "doc": doc}

    # -- reads ---------------------------------------------------------------

    def get(self, key_value: Any, *, conn: sa.Connection | None = None) -> dict | None:
        stmt = sa.select(self.table).where(self.key_col == key_value)
        with self._tx(conn) as c:
            row = c.execute(stmt).one_or_none()
        return None if row is None else self._row_to_record(row)

    def list_all(self, *, conn: sa.Connection | None = None, where=None,
                 order_by=None, limit: int | None = None) -> list[dict]:
        stmt = sa.select(self.table)
        if where is not None:
            stmt = stmt.where(where)
        if order_by is not None:
            stmt = stmt.order_by(order_by)
        if limit is not None:
            stmt = stmt.limit(limit)
        with self._tx(conn) as c:
            rows = c.execute(stmt).all()
        return [self._row_to_record(r) for r in rows]

    def count(self, *, conn: sa.Connection | None = None, where=None) -> int:
        stmt = sa.select(sa.func.count()).select_from(self.table)
        if where is not None:
            stmt = stmt.where(where)
        with self._tx(conn) as c:
            return int(c.execute(stmt).scalar_one())

    # -- writes --------------------------------------------------------------

    def insert(self, record: dict, *, conn: sa.Connection | None = None) -> dict:
        stmt = sa.insert(self.table).values(**self._values(record)).returning(self.table)
        with self._tx(conn) as c:
            row = c.execute(stmt).one()
        return self._row_to_record(row)

    def upsert(self, record: dict, *, conn: sa.Connection | None = None) -> dict:
        values = self._values(record)
        stmt = pg_insert(self.table).values(**values)
        stmt = stmt.on_conflict_do_update(
            index_elements=[self.key_col],
            # doc is REPLACED here, not merged: upsert means "this record is
            # the truth". Use patch() when only some fields are known.
            set_={**{k: v for k, v in values.items() if k != self.key},
                  "updated_at": sa.func.now()},
        ).returning(self.table)
        with self._tx(conn) as c:
            row = c.execute(stmt).one()
        return self._row_to_record(row)

    def patch(self, key_value: Any, changes: dict, *,
              conn: sa.Connection | None = None) -> dict | None:
        """Update only the fields in `changes`, leaving every other field --
        promoted or doc -- exactly as it was."""
        columns, doc_patch = split_doc(changes, self.promoted)
        values: dict[str, Any] = {**columns, "updated_at": sa.func.now()}
        if doc_patch:
            values["doc"] = self.table.c.doc.op("||")(
                sa.cast(sa.literal(doc_patch, sa.JSON), sa.dialects.postgresql.JSONB)
            )
        stmt = (sa.update(self.table)
                .where(self.key_col == key_value)
                .values(**values)
                .returning(self.table))
        with self._tx(conn) as c:
            row = c.execute(stmt).one_or_none()
        return None if row is None else self._row_to_record(row)

    def delete(self, key_value: Any, *, conn: sa.Connection | None = None) -> bool:
        stmt = sa.delete(self.table).where(self.key_col == key_value)
        with self._tx(conn) as c:
            return c.execute(stmt).rowcount > 0
```
